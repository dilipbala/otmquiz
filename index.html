/** * TRIPJACK OTM QUIZ SYSTEM * Simplified Version for Direct Handover */ function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Handle Leaderboard Request if (e.parameter.get === "leaderboard") { var sheet = ss.getSheetByName("Leads"); if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); var data = sheet.getDataRange().getValues(); data.shift(); data.sort((a, b) => b[3] - a[3]); var top10 = data.slice(0, 10).map(row => ({ name: row[0], score: row[3] })); return ContentService.createTextOutput(JSON.stringify(top10)).setMimeType(ContentService.MimeType.JSON); } // Handle Random Questions Request var qSheet = ss.getSheetByName("Questions"); var qData = qSheet.getDataRange().getValues(); qData.shift(); for (var i = qData.length - 1; i > 0; i--) { var j = Math.floor(Math.random() * (i + 1)); var temp = qData[i]; qData[i] = qData[j]; qData[j] = temp; } var selected = qData.slice(0, 5).map(function(row) { var options = [row[2], row[3], row[4], row[5]]; return { q: row[1], a: options, correct: options.indexOf(row[6]), correctText: row[6] }; }); return ContentService.createTextOutput(JSON.stringify(selected)).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Leads"); if (!sheet) { sheet = ss.insertSheet("Leads"); sheet.appendRow(["Name", "Email", "Agency ID", "Score", "Timestamp"]); } var data = JSON.parse(e.postData.contents); sheet.appendRow([data.name, data.email, data.agencyId, data.score, new Date()]); return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); }
